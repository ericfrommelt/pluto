# Lightbox Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a click-to-enlarge lightbox to all image pages with keyboard/button navigation, close on backdrop/Escape, and a per-page opt-out via `data-no-lightbox`.

**Architecture:** A single `Lightbox.astro` component renders the overlay HTML and an inline `<script>` tag with all JS. It's imported once in `Layout.astro` so it's present on every page. No external dependencies.

**Tech Stack:** Astro 5, Tailwind CSS 4, vanilla JS (no framework)

## Task 1: Create the Lightbox component

**Files:**
- Create: `src/components/Lightbox.astro`

- [ ] **Step 1: Create `src/components/Lightbox.astro` with this exact content:**

```astro
---
---

<div id="lightbox" class="hidden fixed inset-0 z-50 flex items-center justify-center" aria-modal="true" role="dialog">
  <!-- Backdrop -->
  <div id="lightbox-backdrop" class="absolute inset-0 bg-black/85"></div>

  <!-- Image -->
  <img
    id="lightbox-img"
    src=""
    alt=""
    class="relative z-10 max-w-[90vw] max-h-[90vh] object-contain"
  />

  <!-- Close button -->
  <button
    id="lightbox-close"
    class="absolute top-4 right-4 z-20 text-white font-mono text-2xl leading-none w-10 h-10 flex items-center justify-center hover:text-neutral-400 transition-colors cursor-pointer"
    aria-label="Close lightbox"
  >&times;</button>

  <!-- Prev button -->
  <button
    id="lightbox-prev"
    class="absolute left-4 top-1/2 -translate-y-1/2 z-20 text-white font-mono text-2xl w-10 h-10 flex items-center justify-center hover:text-neutral-400 transition-colors cursor-pointer"
    aria-label="Previous image"
  >&#8592;</button>

  <!-- Next button -->
  <button
    id="lightbox-next"
    class="absolute right-4 top-1/2 -translate-y-1/2 z-20 text-white font-mono text-2xl w-10 h-10 flex items-center justify-center hover:text-neutral-400 transition-colors cursor-pointer"
    aria-label="Next image"
  >&#8594;</button>

  <!-- Counter -->
  <div
    id="lightbox-counter"
    class="absolute bottom-4 left-1/2 -translate-x-1/2 z-20 text-neutral-400 font-mono text-sm"
  ></div>
</div>

<script>
  document.addEventListener('DOMContentLoaded', () => {
    if ('noLightbox' in document.body.dataset) return;

    const lightbox = document.getElementById('lightbox')!;
    const lightboxImg = document.getElementById('lightbox-img') as HTMLImageElement;
    const lightboxCounter = document.getElementById('lightbox-counter')!;
    const closeBtn = document.getElementById('lightbox-close')!;
    const prevBtn = document.getElementById('lightbox-prev')!;
    const nextBtn = document.getElementById('lightbox-next')!;
    const backdrop = document.getElementById('lightbox-backdrop')!;

    const images: HTMLImageElement[] = Array.from(document.querySelectorAll('img:not(#lightbox-img)'));
    let currentIndex = 0;

    function open(index: number) {
      currentIndex = index;
      const img = images[currentIndex];
      lightboxImg.src = img.src;
      lightboxImg.alt = img.alt;
      updateCounter();
      lightbox.classList.remove('hidden');
      document.addEventListener('keydown', onKeyDown);
    }

    function close() {
      lightbox.classList.add('hidden');
      lightboxImg.src = '';
      document.removeEventListener('keydown', onKeyDown);
    }

    function navigate(delta: number) {
      currentIndex = (currentIndex + delta + images.length) % images.length;
      const img = images[currentIndex];
      lightboxImg.src = img.src;
      lightboxImg.alt = img.alt;
      updateCounter();
    }

    function updateCounter() {
      lightboxCounter.textContent = `${currentIndex + 1} / ${images.length}`;
    }

    function onKeyDown(e: KeyboardEvent) {
      if (e.key === 'Escape') close();
      if (e.key === 'ArrowRight') navigate(1);
      if (e.key === 'ArrowLeft') navigate(-1);
    }

    images.forEach((img, index) => {
      img.style.cursor = 'pointer';
      img.addEventListener('click', () => open(index));
    });

    closeBtn.addEventListener('click', close);
    backdrop.addEventListener('click', close);
    prevBtn.addEventListener('click', () => navigate(-1));
    nextBtn.addEventListener('click', () => navigate(1));
  });
</script>
```

## Task 4: Verify opt-out works

- [ ] **Step 1: Add `data-no-lightbox` to a page body temporarily**

In any page file (e.g. `src/pages/index.astro`), the body tag comes from `Layout.astro`. Since `index.astro` wraps with `<Layout>` and the body is in the layout, add `data-no-lightbox` directly in `Layout.astro`'s `<body>` temporarily:

```astro
<body class="bg-black text-neutral-100" data-no-lightbox>
```

- [ ] **Step 2: Reload the browser** — Click any image. Nothing should happen (no lightbox opens, no cursor change).

- [ ] **Step 3: Remove `data-no-lightbox`** — Revert the `<body>` tag back to `<body class="bg-black text-neutral-100">`. Confirm lightbox works again.
